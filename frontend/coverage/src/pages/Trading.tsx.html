/** Default portfolio: first HOUSE, else first PAPER, else first. */
function pickDefaultPortfolio(pfs: Portfolio[]): string {
  return (
    pfs.find((p) => p.type === "HOUSE")?.portfolio_id ??
    pfs.find((p) => p.type === "PAPER")?.portfolio_id ??
    pfs[0]?.portfolio_id ??
    ""
  );
}
